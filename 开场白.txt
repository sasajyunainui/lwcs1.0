function 克隆JSON值(值, 兜底) {
    if (值 === undefined || 值 === null) return 兜底;

    } catch (错误) {

function 读取原著角色节点年龄(节点) {
    var 匹配 = String(节点 || '').match(/(\d+(?:\.\d+)?)\s*岁/);
    var 年龄 = 匹配 ? Number(匹配[1]) : NaN;
    return Number.isFinite(年龄) ? 年龄 : null;
}

function 读取原著角色快照年龄(快照) {
    var 节点年龄 = 读取原著角色节点年龄(快照 && 快照.节点);
    if (节点年龄 !== null) return 节点年龄;

    return Number.isFinite(年龄) ? 年龄 : 0;
}

function 取原著角色快照(角色记录, 时间线节点) {
    var 快照列表 = Array.isArray(角色记录 && 角色记录.快照) ? 角色记录.快照 : [];
    if (!快照列表.length) return null;
    var 指定快照 = 快照列表.find(function(快照) {
        return String(快照 && 快照.节点 || '').trim() === 时间线节点;
    });
    if (指定快照) return 指定快照;
    var 当前tick = 读取开场tick(时间线节点);
    var 有tick快照 = 快照列表.filter(function(快照) {
        return Number.isFinite(Number(快照 && 快照.tick));
    });
    if (!有tick快照.length) return 快照列表[0];
    var 之前快照 = 有tick快照
        .filter(function(快照) { return Number(快照.tick) <= 当前tick; })
        .sort(function(a, b) { return Number(b.tick) - Number(a.tick); })[0];
    if (之前快照) return 之前快照;
    return 有tick快照.sort(function(a, b) { return Number(a.tick) - Number(b.tick); })[0];
}

function 计算原著角色投影年龄(角色库, 快照, 当前tick) {
    var 快照年龄 = 读取原著角色快照年龄(快照);
    if (快照 && 快照.固定年龄投影 === true) return 快照年龄;
    var 快照tick = Number(快照 && 快照.tick);
    if (!Number.isFinite(快照tick)) return 快照年龄;
    var 每年tick = Math.max(1, Number(角色库 && 角色库.每年tick || 51840));
    return Math.max(0, 快照年龄 + (Number(当前tick || 0) - 快照tick) / 每年tick);
}

function 格式化开场表单文本(值) {
    if (值 === undefined || 值 === null) return '';
    if (typeof 值 === 'string') return 值;
    if (Array.isArray(值)) return 值.map(格式化开场表单文本).filter(Boolean).join('、');
    if (typeof 值 === 'object') {
        return Object.entries(值)
            .map(function(条目) {
                var 子值 = 格式化开场表单文本(条目[1]);
                return 子值 ? 条目[0] + '：' + 子值 : '';
            })
            .filter(Boolean)
            .join('；');
    }
    return String(值);
}

function 刷新原著角色选择() {
    var 选择框 = document.getElementById('original-character-select');

    var 旧值 = 选择框.value || 当前原著角色选择 || '';

    选择框.appendChild(占位);
    if (!角色库) return;
    var 条目列表 = Object.entries(角色库.角色 || {})
        .map(function(条目) {
            var 角色名 = String(条目[0] || '').trim();
            var 角色记录 = 条目[1] || {};
            var 常驻 = Array.isArray(角色记录.开场常驻节点) && 角色记录.开场常驻节点.indexOf(时间线节点) >= 0;
            var 首次登场 = Number(角色记录.首次登场tick);

        })
        .filter(function(条目) { return 条目.角色名; })
        .sort(function(a, b) {
            if (a.常驻 !== b.常驻) return a.常驻 ? -1 : 1;
            return a.首次登场 - b.首次登场 || a.角色名.localeCompare(b.角色名, 'zh-Hans-CN');

        if (首个选项) 选择框.value = 首个选项.value;
    }
    当前原著角色选择 = 选择框.value || '';
}

    var 匹配 = String(生日 || '').trim().match(/^(\d{1,2})\s*月\s*(\d{1,2})\s*日?$/) || String(生日 || '').trim().match(/^(\d{1,2})\s*[-/.]\s*(\d{1,2})$/);
    monthEl.value = String(Math.max(1, Math.min(12, Math.floor(Number(匹配 && 匹配[1]) || 1))));
    dayEl.value = String(Math.max(1, Math.min(30, Math.floor(Number(匹配 && 匹配[2]) || 1))));
}

function 计算开场等级(天赋配置, 年龄值, 生日, 当前tick) {
    var 窗口列表 = [window, window.parent, window.top, 获取开场宿主窗口()];
    for (var 序号 = 0; 序号 < 窗口列表.length; 序号 += 1) {
        try {
            var 计算函数 = 窗口列表[序号] && 窗口列表[序号].__LWCS_CALC_INITIAL_CULTIVATION_LEVEL__;
            if (typeof 计算函数 === 'function') {
                var 等级 = Number(计算函数({
                    天赋梯队: 天赋配置 && 天赋配置.label ? 天赋配置.label : '正常',
                    年龄: 年龄值,
                    底子波动: 1,
                    生日: 生日,
                    当前tick: 当前tick
                }));
                if (Number.isFinite(等级)) return 等级;
            }
        } catch (_) {}
    }
    throw new Error('MVU初始化修为计算入口未就绪');
}

function readStorageJSON(key) {
    try {
        var raw = localStorage.getItem(key);
        if (!raw) return {};
        var parsed = JSON.parse(raw);
        return (parsed && typeof parsed === 'object') ? parsed : {};
    } catch (err) {
        return {};
    }
}

function writeStorageJSON(key, value) {
    try {
        localStorage.setItem(key, JSON.stringify(value || {}));
        return true;
    } catch (err) {
        return false;
    }
}

function applyFormSnapshot(snapshot) {
    if (!snapshot || typeof snapshot !== 'object') return;

    function setValue(id, value) {
        var el = document.getElementById(id);
        if (!el || value === undefined || value === null) return;
        el.value = value;
    }

    setValue('timeline-select', snapshot.timeline);
    setValue('startup-mode', snapshot.开局模式 || '原创角色');
    setValue('char-name', snapshot.name);
    setValue('char-gender', snapshot.gender);
    setValue('char-age', snapshot.age);
    写入开场生日(snapshot.生日);
    setValue('char-loc', snapshot.loc);
    setValue('char-spirit', snapshot.spirit);
    setValue('char-personality', snapshot.personality);
    setValue('char-appearance', snapshot.appearance);
    setValue('char-talent', snapshot.talent);
    setValue('char-bg', snapshot.bg);
    setValue('char-bg-preset', '');
    setValue('char-job-type', snapshot.jobType);
    setValue('char-job-lv', snapshot.jobLv);

    if (typeof window.updateSpiritTypes === 'function') {
        window.updateSpiritTypes();
    }

    var spiritTypes = Array.isArray(snapshot.spiritTypes) ? snapshot.spiritTypes : [];
    var spiritDescs = Array.isArray(snapshot.spiritDescs) ? snapshot.spiritDescs : [];

    document.querySelectorAll('.spirit-type-select').forEach(function(select, index) {
        if (spiritTypes[index] !== undefined) {
            select.value = spiritTypes[index];
        }
    });
    document.querySelectorAll('.spirit-desc-input').forEach(function(input, index) {
        if (spiritDescs[index] !== undefined) {
            input.value = spiritDescs[index];
        }
    });

    var destiny = snapshot.destiny || 'A';
    var selected = document.querySelector('input[name="destiny"][value="' + destiny + '"]');
    if (selected) {
        var card = selected.closest ? selected.closest('.radio-card') : null;
        if (card && typeof window.selectRadio === 'function') {
            window.selectRadio(card);
        } else {
            selected.checked = true;
        }
    }

    if (typeof window.切换开局模式 === 'function') {
        window.切换开局模式(false);
    }
    if (snapshot.开局模式 === '原著角色') {
        刷新原著角色选择();
        setValue('original-character-select', snapshot.原著角色);
        当前原著角色选择 = snapshot.原著角色 || '';
    }
}

function refreshTemplateSelect(selectedName) {
    var select = document.getElementById('template-select');
    if (!select) return;

    var store = loadTemplateStore();
    var names = Object.keys(store).sort(function(a, b) {
        var at = (store[a] && store[a].updatedAt) || 0;
        var bt = (store[b] && store[b].updatedAt) || 0;
        if (bt !== at) return bt - at;
        return a > b ? 1 : (a < b ? -1 : 0);
    });

    select.innerHTML = '';
    var placeholder = document.createElement('option');
    placeholder.value = '';
    placeholder.textContent = '选择已保存模板';
    select.appendChild(placeholder);

    names.forEach(function(name) {
        var option = document.createElement('option');
        option.value = name;
        option.textContent = name;
        select.appendChild(option);
    });

    if (selectedName && store[selectedName]) {
        select.value = selectedName;
    }
}

        select.addEventListener('change', function() {
            var nameInput = document.getElementById('template-name');
            if (nameInput && select.value) {

            }

    var root = document.getElementById('character-creation-root');
    if (root) {
        var persistLast = function() {
            saveLastSnapshot(collectFormSnapshot());
        };
        root.addEventListener('input', persistLast);
        root.addEventListener('change', persistLast);
    }
}

window.saveCurrentTemplate = function() {
    var nameInput = document.getElementById('template-name');
    var name = (nameInput && nameInput.value ? nameInput.value : '').trim();
    if (!name) {
        showTemplateMessage('请先输入模板名，再点击保存。', true);
        return;
    }

    var store = loadTemplateStore();
    store[name] = {
        updatedAt: Date.now(),
        snapshot: collectFormSnapshot()
    };

    if (!saveTemplateStore(store)) {
        showTemplateMessage('保存失败：本地存储不可用。', true);
        return;
    }

    refreshTemplateSelect(name);
    showTemplateMessage('已保存模板：' + name);
};

window.loadSelectedTemplate = function() {
    var select = document.getElementById('template-select');
    var name = (select && select.value) ? select.value : '';
    if (!name) {
        showTemplateMessage('请先从列表选择模板。', true);
        return;
    }

    var store = loadTemplateStore();
    var entry = store[name];
    if (!entry || !entry.snapshot) {
        showTemplateMessage('模板不存在，列表已刷新。', true);
        refreshTemplateSelect();
        return;
    }

    applyFormSnapshot(entry.snapshot);
    saveLastSnapshot(entry.snapshot);

    var nameInput = document.getElementById('template-name');
    if (nameInput) nameInput.value = name;

    showTemplateMessage('已加载模板：' + name);
};

window.deleteSelectedTemplate = function() {
    var select = document.getElementById('template-select');
    var name = (select && select.value) ? select.value : '';
    if (!name) {
        showTemplateMessage('请先选择要删除的模板。', true);
        return;
    }

    var store = loadTemplateStore();
    if (!store[name]) {
        showTemplateMessage('模板不存在，列表已刷新。', true);
        refreshTemplateSelect();
        return;
    }

    delete store[name];
    if (!saveTemplateStore(store)) {
        showTemplateMessage('删除失败：本地存储不可用。', true);
        return;
    }

    refreshTemplateSelect();
    showTemplateMessage('已删除模板：' + name);
};
